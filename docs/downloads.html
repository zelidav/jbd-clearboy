Rebuild from source

Python 3.12
# CadQuery has no 3.13/3.14 wheels yet
py install 3.12
python3.12 -m venv cadenv && ./cadenv/Scripts/python -m pip install -r requirements.txt

./cadenv/Scripts/python cad/model.py out      # hammer solid -> STEP + STL
./cadenv/Scripts/python cad/frit.py out       # frit grains + marbles
./cadenv/Scripts/python cad/jar.py out        # jar, its frit and its marbles
./cadenv/Scripts/python cad/glb_export.py     # GLB with glass materials
./cadenv/Scripts/python cad/turntable.py      # 72-frame turntables
./cadenv/Scripts/python cad/encode.py         # mp4 loops + spinner frames
./cadenv/Scripts/python cad/build_web.py      # this site
